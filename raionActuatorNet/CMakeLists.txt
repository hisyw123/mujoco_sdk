cmake_minimum_required(VERSION 3.18)
project(raion_actuator_net LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MUJOCO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Existing MuJoCo source")
set(MUJOCO_LIBRARY "${MUJOCO_SOURCE}/build-greeny/lib/libmujoco.so" CACHE FILEPATH "Existing native MuJoCo library")
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(glfw3 REQUIRED)
find_package(OpenGL REQUIRED)
add_executable(augmented_sim src/augmented_sim.cpp)
target_include_directories(augmented_sim PRIVATE "${MUJOCO_SOURCE}/include")
target_link_libraries(augmented_sim PRIVATE "${MUJOCO_LIBRARY}" Eigen3::Eigen yaml-cpp glfw OpenGL::GL)
get_filename_component(MUJOCO_LIBDIR "${MUJOCO_LIBRARY}" DIRECTORY)
set_target_properties(augmented_sim PROPERTIES BUILD_RPATH "${MUJOCO_LIBDIR}")
target_compile_options(augmented_sim PRIVATE -Wall -Wextra)
